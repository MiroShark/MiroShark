# Seed Chat Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace MiroShark's Home page with a Q&A chat that uses Claude (Max-plan CLI) to slot-fill a structured seed, then launch the existing research → project → Decision Lab pipeline.

**Architecture:** New Flask blueprint `/api/seed-chat` with two endpoints (`/turn`, `/launch`). One LLM call per turn returns assistant reply + updated slot state + readiness flag in a single JSON envelope. New Vue front-page (`SeedChat.vue`) with a chat pane and a live slots sidebar. Old Home preserved at `/legacy`.

**Tech Stack:** Flask, Vue 3, axios, `claude -p` CLI via existing `ClaudeCodeClient`, pytest (newly added for backend), Neo4j (unchanged), Ollama embeddings (unchanged).

**Spec:** [`docs/superpowers/specs/2026-04-28-seed-chat-design.md`](../specs/2026-04-28-seed-chat-design.md)

```python
"""
Seed extractor: drives one turn of the seed-chat conversation.

Each call sends the full message history + current slot state to the LLM and
expects a single JSON envelope: {reply, slots, ready}. The backend re-checks
required slots so a hallucinated `ready=True` from the LLM cannot launch
an under-specified pipeline.
"""

import subprocess
from typing import Dict, List, Tuple

from ..utils.llm_client import create_llm_client
from ..utils.logger import get_logger

logger = get_logger("miroshark.seed_extractor")

REQUIRED_SLOTS = ("topic", "intent", "output_format")
REQUIRED_LIST_SLOTS = ("stakeholders",)
MIN_STAKEHOLDERS = 2

EMPTY_STATE: Dict = {
    "topic": "",
    "intent": "",
    "stakeholders": [],
    "decision_branches": [],
    "contested_claims": [],
    "output_format": "",
}

SYSTEM_PROMPT = """You are MiroShark's seed-chat assistant. Your job is to help the user
turn a vague research question into a structured "seed" that drives a downstream
research and simulation pipeline.

You must return a single JSON object with exactly these keys:

{
  "reply": "string — your next message to the user",
  "slots": {
    "topic": "string — the issue/question being investigated",
    "intent": "string — what the user wants out of this (e.g., pros/cons brief, decision memo)",
    "stakeholders": [{"name": "string", "role": "string", "stance": "supporting|opposing|neutral|unknown"}],
    "decision_branches": [{"label": "string", "description": "string"}],
    "contested_claims": ["string — a claim worth investigating"],
    "output_format": "pros_cons | decision_memo | executive_summary | full_report"
  },
  "ready": true | false
}

Rules:
- Required slots: topic, intent, stakeholders (at least 2), output_format.
- decision_branches and contested_claims are optional. Leave empty arrays if not applicable. Do NOT invent.
- Each turn, re-emit the FULL slots object — fill what you can, preserve what was already there.
- Set ready=true only when required slots are populated AND the user has confirmed they're done refining.
- Ask one targeted question per turn, prioritising the most consequential missing slot.
- Be concise. The reply field is shown verbatim to the user.

If the user pushes back on a suggestion, accept their version and update slots accordingly.
"""


def _required_slots_filled(slots: Dict) -> bool:
    for key in REQUIRED_SLOTS:
        if not slots.get(key):
            return False
    if len(slots.get("stakeholders", [])) < MIN_STAKEHOLDERS:
        return False
    return True


def _normalise_slots(slots: Dict) -> Dict:
    """Ensure all expected keys exist; fill missing with empty defaults."""
    normalised = dict(EMPTY_STATE)
    for key in EMPTY_STATE:
        if key in slots:
            normalised[key] = slots[key]
    return normalised


def process_turn(
    messages: List[Dict[str, str]],
    current_state: Dict,
) -> Tuple[str, Dict, bool]:
    """
    Run one Q&A turn.

    Args:
        messages: full chat history as [{"role": "user"|"assistant", "content": str}, ...]
        current_state: prior slot state (so the LLM can re-emit it intact)

    Returns:
        (assistant_reply, updated_slots, ready_to_launch)
    """
    llm = create_llm_client()

    state_summary = (
        "Current slot state (re-emit and update):\n"
        + str(current_state)
    )
    full_messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "system", "content": state_summary},
        *messages,
    ]

    try:
        envelope = llm.chat_json(messages=full_messages)
    except (ValueError, KeyError, TypeError) as exc:
        logger.warning("seed_extractor: malformed LLM response (%s); preserving state", exc)
        return (
            "Sorry — I had trouble parsing my own response. Could you say that again, or rephrase?",
            current_state,
            False,
        )
    except subprocess.TimeoutExpired as exc:
        logger.error("seed_extractor: Claude CLI timed out (%s)", exc)
        raise RuntimeError("claude_cli_timeout") from exc

    reply = envelope.get("reply", "").strip()
    slots = _normalise_slots(envelope.get("slots", {}))
    llm_ready = bool(envelope.get("ready", False))

    ready = llm_ready and _required_slots_filled(slots)

    if not reply:
        reply = "(no response — please continue)"

    return reply, slots, ready
```

```python
"""Seed-chat API routes."""

from flask import jsonify, request

from . import seed_chat_bp
from ..services.seed_extractor import process_turn, EMPTY_STATE
from ..utils.logger import get_logger

logger = get_logger("miroshark.api.seed_chat")


@seed_chat_bp.route("/turn", methods=["POST"])
def turn():
    payload = request.get_json(silent=True) or {}
    messages = payload.get("messages")
    seed_state = payload.get("seed_state") or dict(EMPTY_STATE)

    if not isinstance(messages, list) or not messages:
        return jsonify({"error": "messages must be a non-empty list"}), 400

    try:
        reply, updated, ready = process_turn(messages, seed_state)
    except RuntimeError as exc:
        logger.error("seed-chat turn failed: %s", exc)
        return jsonify({"error": f"claude_cli_unavailable: {exc}"}), 503

    return jsonify({
        "assistant_message": reply,
        "updated_seed_state": updated,
        "ready_to_launch": ready,
    })
```

```python
"""
Seed launcher: maps a complete seed onto existing pipeline calls.

Phase B v1:
  - Creates a project named after the topic.
  - Synthesises the document content from intent + topic + claims +
    research-agent findings.
  - Optionally creates a Decision Lab if decision_branches is non-empty.
  - Returns {project_id, lab_id?} for the frontend to redirect.
"""

from typing import Dict, List, Optional

from ..models.project import ProjectManager
from ..services.research_agent import research_with_intent
from ..utils.logger import get_logger

logger = get_logger("miroshark.seed_launcher")

REQUIRED_KEYS = ("topic", "intent", "output_format")
MIN_STAKEHOLDERS = 2


def _validate(seed: Dict) -> None:
    for key in REQUIRED_KEYS:
        if not seed.get(key):
            raise ValueError(f"seed missing required field: {key}")
    if len(seed.get("stakeholders", [])) < MIN_STAKEHOLDERS:
        raise ValueError(f"seed required field: stakeholders (need at least {MIN_STAKEHOLDERS})")


def _build_seed_document(seed: Dict, research_summary: Dict) -> str:
    parts = [
        f"# {seed['topic']}",
        "",
        f"**Intent:** {seed['intent']}",
        "",
        "## Stakeholders",
    ]
    for s in seed["stakeholders"]:
        parts.append(f"- **{s.get('name', '?')}** ({s.get('role', '?')}) — stance: {s.get('stance', 'unknown')}")

    if seed.get("contested_claims"):
        parts += ["", "## Contested claims"]
        for c in seed["contested_claims"]:
            parts.append(f"- {c}")

    if seed.get("decision_branches"):
        parts += ["", "## Decision branches"]
        for b in seed["decision_branches"]:
            parts.append(f"- **{b.get('label', '?')}** — {b.get('description', '')}")

    parts += ["", "## Research summary", str(research_summary)]
    return "\n".join(parts)


def create_lab_with_branches(topic: str, branches: List[Dict]) -> Optional[Dict]:
    """Create a Decision Lab seeded with the given branches.

    Returns {"lab_id": str} or None if not implementable in v1.
    Real implementation calls DecisionLabManager directly — to be wired
    against the actual class discovered in step 1 of the implementing
    task. Until then, this is the stub for tests to mock.
    """
    raise NotImplementedError(
        "Wire to DecisionLabManager.create_lab + add_branch using the helpers "
        "discovered in backend/app/services/decision_lab*.py."
    )


def launch_seed(seed: Dict) -> Dict:
    _validate(seed)

    project = ProjectManager.create_project(name=seed["topic"])
    project.simulation_requirement = seed["intent"]

    research_intent = (
        f"{seed['intent']}\n\n"
        f"Topic: {seed['topic']}\n"
        f"Contested claims to investigate:\n"
        + "\n".join(f"- {c}" for c in seed.get("contested_claims", []))
    )
    report = research_with_intent(content="", intent=research_intent)
    summary = report.to_dict() if hasattr(report, "to_dict") else {}

    seed_document = _build_seed_document(seed, summary)
    project.files.append({"filename": "seed.md", "size": len(seed_document)})

    result = {"project_id": project.project_id}

    if seed.get("decision_branches"):
        lab = create_lab_with_branches(seed["topic"], seed["decision_branches"])
        if lab and lab.get("lab_id"):
            result["lab_id"] = lab["lab_id"]

    logger.info("seed launched: project=%s lab=%s", result.get("project_id"), result.get("lab_id"))
    return result
```
